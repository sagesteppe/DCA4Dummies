Decision Curve Analysis for Dummies

An interactive companion to the lsed methods

Author

sagesteppe

Published

November 28, 2026

A rare plant you manage has 8 known populations; fall short of 12 and it gets listed, costing roughly $60,000. A species distribution model (SDM) scores all 500 candidate sites for occupancy — but the plant is rare, only about 2% of sites are occupied, and a field crew can visit perhaps 30. Where do you send them?

Decision curve analysis (DCA) answers that in one currency — net benefit — and the lsed package rebuilds the whole curve from a single reported score. The three panels below are the method in miniature; full derivations and validation are in the manuscript and the appendices.

The threshold probability

How sure must the model be before a site is worth a crew-day? That cutoff, \(p_t\), encodes your costs — not a statistic:

\[\frac{p_t}{1 - p_t} = \frac{\text{cost of a wasted survey}}{\text{value of an occupied site found}}\]

DCA evaluates every \(p_t\) at once. For any strategy, at threshold \(p_t\):

\[\text{Net benefit} = \frac{TP}{n} \;-\; \frac{FP}{n}\cdot\frac{p_t}{1 - p_t}\]

“Survey nothing” always scores 0 — the baseline a model has to beat.

normPDF = (x) => Math.exp(-0.5 * x * x) / Math.sqrt(2 * Math.PI)

// standard normal CDF (Zelen & Severo)
function normCDF(x) {
  const t = 1 / (1 + 0.2316419 * Math.abs(x));
  const d = normPDF(x);
  const p = d * t * (0.319381530 + t * (-0.356563782 + t * (1.781477937 +
            t * (-1.821255978 + t * 1.330274429))));
  return x >= 0 ? 1 - p : p;
}

// standard normal quantile (Acklam)
function normInv(p) {
  const a = [-3.969683028665376e+01, 2.209460984245205e+02, -2.759285104469687e+02,
             1.383577518672690e+02, -3.066479806614716e+01, 2.506628277459239e+00];
  const b = [-5.447609879822406e+01, 1.615858368580409e+02, -1.556989798598866e+02,
             6.680131188771972e+01, -1.328068155288572e+01];
  const c = [-7.784894002430293e-03, -3.223964580411365e-01, -2.400758277161838e+00,
             -2.549732539343734e+00, 4.374664141464968e+00, 2.938163982698783e+00];
  const d = [7.784695709041462e-03, 3.224671290700398e-01, 2.445134137142996e+00,
             3.754408661907416e+00];
  const pl = 0.02425, ph = 1 - pl;
  let q, r;
  if (p < pl) { q = Math.sqrt(-2 * Math.log(p));
    return (((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
           ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1); }
  if (p <= ph) { q = p - 0.5; r = q*q;
    return (((((a[0]*r+a[1])*r+a[2])*r+a[3])*r+a[4])*r+a[5])*q /
           (((((b[0]*r+b[1])*r+b[2])*r+b[3])*r+b[4])*r+1); }
  q = Math.sqrt(-2 * Math.log(1 - p));
  return -(((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
          ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1);
}

// ap_binormal(): average precision of an equal-variance binormal ROC at prevalence pi
function apBinormal(mu, pi, grid = 1500) {
  let rPrev = 0, pPrev = 1, area = 0;
  for (let i = 0; i < grid; i++) {
    const f = 1e-6 + (1 - 2e-6) * i / (grid - 1);     // FPR ascending
    const tpr = normCDF(normInv(f) + mu);             // recall (prevalence-invariant)
    const prec = (pi * tpr) / (pi * tpr + (1 - pi) * f);
    area += (tpr - rPrev) * (prec + pPrev) / 2;        // trapezoid
    rPrev = tpr; pPrev = prec;
  }
  return area;
}

// invert a reported PR-AUC at design prevalence pi0 -> binormal separation mu
function fitMu(prauc, pi0) {
  let lo = 1e-3, hi = 8;
  if (prauc <= apBinormal(lo, pi0)) return lo;
  if (prauc >= apBinormal(hi, pi0)) return hi;
  for (let k = 0; k < 44; k++) {
    const mid = (lo + hi) / 2;
    if (apBinormal(mid, pi0) < prauc) lo = mid; else hi = mid;
  }
  return (lo + hi) / 2;
}

// nb_binormal(): best-achievable net benefit from the binormal ROC at prevalence pi
function nbBinormal(mu, pi, t) {
  const m = ((1 - pi) / pi) * (t / (1 - t));     // target ROC slope
  const cc = (Math.log(m) + mu * mu / 2) / mu;   // score cutoff giving that slope
  return pi * normCDF(mu - cc) - (1 - pi) * normCDF(-cc) * (t / (1 - t));
}

// nb_treat_all(): the "survey every site" reference
nbTreatAll = (pi, t) => pi - (1 - pi) * (t / (1 - t))

// P(X >= k) for X ~ Poisson(lambda)
function poissonSF(k, lam) {
  if (k <= 0) return 1;
  let term = Math.exp(-lam), cdf = term;
  for (let i = 1; i < k; i++) { term *= lam / i; cdf += term; }
  return 1 - cdf;
}

money = (v) => (v < 0 ? "−$" : "+$") + Math.abs(Math.round(v)).toLocaleString()
M = 500          // candidate sites scored by the SDM
KNOWN = 8        // populations known now
TARGET = 12      // populations needed to avoid listing
NEED = TARGET - KNOWN   // = 4 more
CREW = 6400

1 · From a reported score to a decision curve

A model reports one number on its test set — a PR-AUC (average precision). Fitting an equal-variance binormal ROC to that single scalar (ap_binormal, nb_binormal) reconstructs the full decision curve — no raw predictions required.

viewof prauc1 = Inputs.range([0.10, 0.95], { value: 0.70, step: 0.01, label: "Reported PR-AUC (on the test set)" })
viewof pi0 = Inputs.range([0.05, 0.50], { value: 0.50, step: 0.01, label: "Evaluation-set prevalence (π₀)" })
mu1 = fitMu(prauc1, pi0)
impliedROC1 = normCDF(mu1 / Math.sqrt(2))     // implied_roc_auc = pnorm(mu / sqrt2)
ptGrid1 = d3.range(0.02, 0.8001, 0.01)
nbEval = ptGrid1.flatMap(t => [
  { pt: t, nb: nbBinormal(mu1, pi0, t), strategy: "Use the model" },
  { pt: t, nb: nbTreatAll(pi0, t),      strategy: "Survey every site" }
])
nbPeak1 = d3.max(ptGrid1, t => nbBinormal(mu1, pi0, t))

2 · Re-price the curve to landscape prevalence

A test set is usually balanced; the landscape is not. Holding the fitted ROC fixed and re-pricing precision to the true prevalence \(\pi\) (reweight, ap_scalar_to_prev) shows what the model is worth where it will actually be used. At 2% occupancy the “survey everything” line collapses and net benefits shrink by an order of magnitude.

viewof piL = Inputs.range([0.005, 0.20], { value: 0.02, step: 0.005, label: "Landscape prevalence (π)" })
apEval1  = apBinormal(mu1, pi0)      // PR-AUC at the evaluation prevalence
apLand   = apBinormal(mu1, piL)      // PR-AUC re-priced to the landscape
ptGrid2  = d3.range(0.02, 0.8001, 0.01)
nbLand = ptGrid2.flatMap(t => [
  { pt: t, nb: nbBinormal(mu1, piL, t), strategy: "Use the model" },
  { pt: t, nb: nbTreatAll(piL, t),      strategy: "Survey every site" }
])
nbPeakL = d3.max(ptGrid2, t => nbBinormal(mu1, piL, t))
dcaLand = Plot.plot({
  width: 640, height: 360, marginLeft: 64, marginBottom: 46,
  style: { fontSize: "12px", background: "transparent", color: "#c3c2b7" },
  title: `Same model, re-priced to π = ${(piL*100).toFixed(1)}%`,

  y: { label: "↑ Net benefit", domain: [Math.min(-0.01, -nbPeakL * 0.6), Math.max(0.01, nbPeakL * 1.2)], grid: true },

3 · Cost–benefit: what the curve buys you

Net benefit per site is tiny at 2% prevalence (you just saw ~0.006), which feels abstract. Set your actual costs — the threshold \(p_t\) follows from them — and the curve translates into populations found and dollars.

viewof cwaste = Inputs.range([50, 1000], { value: 200, step: 25, label: "Cost of one wasted survey ($/site)" })
viewof listcost = Inputs.range([20000, 150000], { value: 60000, step: 5000, label: "Listing cost if < 12 populations ($)" })
viewof capacity = Inputs.range([10, 80], { value: 30, step: 5, label: "Crew capacity (sites surveyable)" })
vFound = listcost / NEED
pt3 = cwaste / (cwaste + vFound)                 // textbook DCA threshold from costs

// score cutoff where posterior(c) == pt3, under the landscape binormal
posterior3 = (x) => {
  const fP = piL * normPDF(x - mu1), fA = (1 - piL) * normPDF(x);
  return fP / (fP + fA);
}
fracFlag = (c) => piL * (1 - normCDF(c - mu1)) + (1 - piL) * (1 - normCDF(c))
function solveCut(predicate) {        // bisect for a score cutoff, c increasing
  let lo = -12, hi = 18;
  for (let k = 0; k < 60; k++) { const mid = (lo + hi) / 2; if (predicate(mid)) hi = mid; else lo = mid; }
  return (lo + hi) / 2;
}
cPt  = solveCut(c => posterior3(c) >= pt3)          // cutoff from the cost threshold
cCap = solveCut(c => M * fracFlag(c) <= capacity)   // cutoff from crew capacity
cEff = Math.max(cPt, cCap)                          // whichever is more selective
nFlag3 = Math.round(M * fracFlag(cPt))
nSurvey3 = Math.min(nFlag3, capacity)
finds3 = M * piL * (1 - normCDF(cEff - mu1))        // expected occupied sites found
pwin3 = poissonSF(NEED, finds3)                     // ≈ P(find the 4 you need), Poisson(finds)
dollars3 = pwin3 * listcost - CREW                  // expected savings vs. doing nothing
nbAtPt = nbBinormal(mu1, piL, pt3)
htl.html`<div class="readouts">
  <div class="tile"><div class="label">Cost-derived threshold pₜ</div><div class="value">${(pt3*100).toFixed(0)}%</div><div class="sub">net benefit here: ${nbAtPt.toFixed(4)}</div></div>
  <div class="tile"><div class="label">Sites flagged → surveyed</div><div class="value">${nFlag3}<span style="font-size:0.9rem;font-weight:400;color:#898781"> → ${nSurvey3}</span></div><div class="sub">crew capacity ${capacity}</div></div>
  <div class="tile"><div class="label">Expected populations found</div><div class="value ${finds3 >= NEED ? "pos" : ""}">${finds3.toFixed(1)}</div><div class="sub">you need ${NEED} (${KNOWN} → ${TARGET})</div></div>
  <div class="tile"><div class="label">P(avoid listing) ≈</div><div class="value ${pwin3 >= 0.5 ? "pos" : (pwin3 < 0.2 ? "neg" : "")}">${(pwin3*100).toFixed(0)}%</div><div class="sub">Poisson(expected finds)</div></div>
  <div class="tile"><div class="label">Expected \$ vs. doing nothing</div><div class="value ${dollars3 > 0 ? "pos" : "neg"}">${money(dollars3)}</div><div class="sub">P·listing − crew cost</div></div>
</div>`
{
  let cls, msg;
  const idle = capacity - nSurvey3;
  if (pwin3 >= 0.6) {
    cls = "win";
    msg = html`Your costs set <b>pₜ = ${(pt3*100).toFixed(0)}%</b>. Surveying the model's
      top ${nSurvey3} sites, you expect <b>${finds3.toFixed(1)} of the ${NEED}</b> populations
      you need — a <b>${(pwin3*100).toFixed(0)}%</b> chance of staying off the list, worth
      <b>${money(dollars3)}</b>.`;
  } else if (pwin3 >= 0.3) {
    cls = "tie";
    msg = html`At <b>pₜ = ${(pt3*100).toFixed(0)}%</b> you've a <b>${(pwin3*100).toFixed(0)}%</b>
      shot. ${idle > 1 ? html`You're flagging only ${nFlag3} sites and leaving <b>${idle} crew-days
      idle</b> — the crew is paid for the season, so lower the wasted-survey cost (and thus pₜ)
      to put them all to work.` : html`A stronger model or more capacity moves the needle most here.`}`;
  } else {
    cls = "lose";
    msg = html`At <b>pₜ = ${(pt3*100).toFixed(0)}%</b> the odds of clearing the bar are only
      <b>${(pwin3*100).toFixed(0)}%</b>. ${idle > 1 ? html`You're using just ${nSurvey3} of
      ${capacity} crew-days — lower pₜ to use the rest.` : html`This model and crew can't reliably
      reach ${TARGET}; a better SDM or a bigger crew is the fix.`}`;
  }
  return htl.html`<div class="verdict ${cls}">${msg}</div>`;
}

Everything runs in your browser. The decision curves are exact (nb_binormal); the P(avoid listing) tile is a Poisson approximation to the count of occupied sites among those surveyed — the manuscript uses the full simulation.
